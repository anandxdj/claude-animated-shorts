(window as any).drawSheet = (key: string) => {
  const C = CASTS[key], n = C.cards.length, cw = 300, ch = 520, pad = 20, head = 70;
  const cv = document.createElement("canvas"); cv.width = n * cw + pad * 2; cv.height = ch + head + 50; document.body.appendChild(cv);
  const g = cv.getContext("2d")!;
  g.fillStyle = C.bg; g.fillRect(0, 0, cv.width, cv.height);
  g.fillStyle = "#241726"; g.font = "bold 30px sans-serif"; g.fillText(C.title, pad, 46);
  C.cards.forEach((c, i) => {
    g.save(); g.translate(pad + i * cw, head); g.scale(cw / 600, cw / 600);
    g.fillStyle = "rgba(36,23,38,0.18)"; g.beginPath(); g.ellipse(300, 975, 120, 18, 0, 0, Math.PI * 2); g.fill();
    const p = new Pen(g);
    bodyFront(p, c.o, { expr: c.expr, mode: c.mode ?? "stand", phase: c.phase ?? 0.1, seed: c.seed, facing: c.facing ?? 0, tilt: c.tilt ?? 0, headTilt: c.headTilt ?? 0, prop: c.prop ?? null });
    g.restore();
    g.fillStyle = "#241726"; g.font = "20px sans-serif"; g.textAlign = "center";
    g.fillText(c.name, pad + i * cw + cw / 2, head + ch + 30); g.textAlign = "left";
  });
  return cv.toDataURL("image/png");
};
